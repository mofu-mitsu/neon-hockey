// ------------------------------------------------------------------------------
// 設定項目
// ------------------------------------------------------------------------------
// シート名の指定（存在しない場合は自動作成されます）
const SHEET_NAME = 'プレイログ';

// スプレッドシートID（通常は空文字 '' のままでOK）
// ※スプレッドシート内の「拡張機能」から開いた場合は自動取得されるため空文字で動きます。
// ※もし script.google.com からスタンドアロンで作成した場合は、スプレッドシートのURLに含まれるIDを貼り付けてください。
const SPREADSHEET_ID = '';

// 列ヘッダー定義
const HEADERS = [
  '日時',
  'ゲームモード',
  '対戦形式',
  '勝敗結果',
  'プレイヤー得点',
  '相手得点',
  'プレイヤー名',
  '味方NPC',
  '対戦相手1',
  '対戦相手2',
  '試合時間(秒)',
  '最長ラリー',
  '強打スマッシュ',
  'ターゲット/ストーン',
  '利用端末'
];

/**
 * 対象のスプレッドシートを取得するヘルパー
 */
function getTargetSpreadsheet() {
  if (SPREADSHEET_ID && SPREADSHEET_ID.trim() !== '') {
    return SpreadsheetApp.openById(SPREADSHEET_ID.trim());
  }
  const active = SpreadsheetApp.getActiveSpreadsheet();
  if (!active) {
    throw new Error('アクティブなスプレッドシートが見つかりません。SPREADSHEET_ID にスプレッドシートのIDを指定してください。');
  }
  return active;
}

/**
 * データをシートに1行追加する共通関数
 */
function appendRecordToSheet(data) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(10000); // 同時アクセス時の競合を防ぐため最大10秒待機
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: 'error',
      message: 'Server busy, lock timeout'
    })).setMimeType(ContentService.MimeType.JSON);
  }

  try {
    const ss = getTargetSpreadsheet();
    let sheet = ss.getSheetByName(SHEET_NAME);
    if (!sheet) {
      sheet = ss.insertSheet(SHEET_NAME);
    }

    // シートが空ならヘッダー行を書き込み＆装飾
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(HEADERS);
      const headerRange = sheet.getRange(1, 1, 1, HEADERS.length);
      headerRange.setBackground('#0f172a'); // サイバーダーク
      headerRange.setFontColor('#38bdf8'); // ネオンシアン
      headerRange.setFontWeight('bold');
      headerRange.setHorizontalAlignment('center');
      sheet.setFrozenRows(1);
    }

    // データの組み立て
    const row = [
      data.timestamp || Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy/MM/dd HH:mm:ss'),
      data.mode || '—',
      data.teamFormat || '—',
      data.result || '—',
      data.playerGoals !== undefined ? data.playerGoals : 0,
      data.opponentGoals !== undefined ? data.opponentGoals : 0,
      data.playerName || 'あなた',
      data.allyName || '—',
      data.opponent1Name || '—',
      data.opponent2Name || '—',
      data.matchDurationSec || 0,
      data.longestRally || 0,
      data.superSmashes || 0,
      data.targetsHit || 0,
      data.device || '—'
    ];

    sheet.appendRow(row);

    return ContentService.createTextOutput(JSON.stringify({
      status: 'success',
      message: 'Logged match data successfully'
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({
      status: 'error',
      message: error.toString()
    })).setMimeType(ContentService.MimeType.JSON);

  } finally {
    lock.releaseLock();
  }
}

/**
 * WebアプリへPOSTリクエストが届いた時に実行される関数
 */
function doPost(e) {
  try {
    let data;
    if (e && e.postData && e.postData.contents) {
      data = JSON.parse(e.postData.contents);
    } else if (e && e.parameter) {
      data = e.parameter;
    } else {
      data = {};
    }
    return appendRecordToSheet(data);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: 'error',
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

/**
 * ブラウザでURLを直接開いた時やGETビーコンを受信した時の関数
 */
function doGet(e) {
  // GETパラメータ経由のロギング対応
  if (e && e.parameter && e.parameter.payload) {
    try {
      const data = JSON.parse(e.parameter.payload);
      return appendRecordToSheet(data);
    } catch (err) {
      return ContentService.createTextOutput(JSON.stringify({
        status: 'error',
        message: err.toString()
      })).setMimeType(ContentService.MimeType.JSON);
    }
  }

  return ContentService.createTextOutput(
    '🎮 NEON HOCKEY: CHARACTER ARENA プレイログ記録用 GAS Web App は正常に稼働しています！'
  ).setMimeType(ContentService.MimeType.TEXT);
}
